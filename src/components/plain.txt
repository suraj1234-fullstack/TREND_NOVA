import React, { useState, useEffect } from 'react';
import { useNavigate, useLocation } from 'react-router-dom';
import AdminDashboard from '../adminDashboard';
import { Link } from 'react-router-dom';

const Navbar = ({ darkMode, setDarkMode }) => {
  const [isOpen, setIsOpen] = useState(false);
  const [activeDropdown, setActiveDropdown] = useState(null);
  const [activeTab, setActiveTab] = useState('categories'); // 'categories' or 'brands'
  const [searchQuery, setSearchQuery] = useState('');
  
  // --- VOICE SEARCH STATE ---
  const [isListening, setIsListening] = useState(false);
  
  // --- SCROLL SHRINK STATE FOR SUB-BAR ---
  const [isScrolled, setIsScrolled] = useState(false);

  // --- DYNAMIC SCROLL BACKGROUND COLOR LOGIC ---
  const [navBg, setNavBg] = useState('#ffffff');
  const navigate = useNavigate();
  const location = useLocation();

  const handleGoogleSignIn = () => {
    window.location.href = 'https://auth-api-gtqf.onrender.com/api/auth/google';
  };

  const handleLogout = () => {
    localStorage.clear();
    window.location.href = '/';
  };

  const [token, setToken] = useState(localStorage.getItem('token'));
  const [role, setRole] = useState(localStorage.getItem('role'));

  useEffect(() => {
    const handleStorage = () => {
      setToken(localStorage.getItem('token'));
      setRole(localStorage.getItem('role'));
    };
    window.addEventListener('storage', handleStorage);
    return () => window.removeEventListener('storage', handleStorage);
  }, []);

  // Dynamic Theme Body Sync
  useEffect(() => {
    if (darkMode) {
      document.body.classList.add('dark');
      document.documentElement.classList.add('dark');
    } else {
      document.body.classList.remove('dark');
      document.documentElement.classList.remove('dark');
    }
  }, [darkMode]);

  useEffect(() => {
    const handleScroll = () => {
      const scrollY = window.scrollY;

      if (scrollY > 40) {
        setIsScrolled(true);
      } else {
        setIsScrolled(false);
      }

      if (scrollY > 2400) {
        setNavBg('#fbcfe8');
      } else if (scrollY > 1900) {
        setNavBg('#fef6aa');
      } else if (scrollY > 1400) {
        setNavBg('#fef08a');
      } else if (scrollY > 900) {
        setNavBg('#bbf7d0');
      } else if (scrollY > 400) {
        setNavBg('#bfdbfe');
      } else if (scrollY > 100) {
        setNavBg('#e2e8f0');
      } else {
        setNavBg('#ffffff');
      }
    };

    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  const subCategoriesList = [
    { title: 'Fashion', icon: <i className="fa-solid fa-vest"></i>, link: '/search?q=fashion' },
    { title: 'Mobiles', icon: <i className="fa-solid fa-mobile-screen-button"></i>, link: '/search?q=mobiles' },
    { title: 'Electronics', icon: <i className="fa-solid fa-tv"></i>, link: '/search?q=electronics' },
    { title: 'Beauty', icon: <i className="fa-solid fa-venus-double"></i>, link: '/search?q=beauty' },
    { title: 'Home', icon: <i className="fa-solid fa-house-chimney"></i>, link: '/search?q=home' },
    { title: 'Appliances', icon:<i className="fa-solid fa-bolt"></i>, link: '/search?q=appliances' },
    { title: 'Toys', icon: <i className="fa-solid fa-cubes-stacked"></i>, link: '/search?q=toys' },
    { title: 'Grocery', icon: <i className="fa-solid fa-basket-shopping"></i>, link: '/search?q=grocery' },
    { title: 'Two Wheelers', icon:<i className="fa-solid fa-motorcycle"></i>, link: '/search?q=two-wheelers' }
  ];

  const categoriesData = {
    Men: {
      categories: [
        { title: "Topwear", items: ["T-Shirts", "Casual Shirts", "Formal Shirts", "Polos", "Hoodies", "Jackets"] },
        { title: "Bottomwear", items: ["Jeans", "Trousers", "Track Pants", "Shorts", "Joggers"] },
        { title: "Footwear", items: ["Casual Shoes", "Sports Shoes", "Formal Shoes", "Sneakers", "Sandals"] }
      ],
      brands: [
        { title: "Top Clothing Brands", items: ["Nike", "Adidas", "Puma", "Levi's", "Zara", "H&M"] }
      ]
    },
    Women: {
      categories: [
        { title: "Indian & Fusion Wear", items: ["Kurtas & Suits", "Sarees", "Lehengas", "Ethnic Sets"] },
        { title: "Western Wear", items: ["Dresses", "Tops", "Jeans", "Skirts", "Jumpsuits"] },
        { title: "Footwear", items: ["Flats", "Heels", "Sneakers", "Boots"] }
      ],
      brands: [
        { title: "Ethnic Brands", items: ["Biba", "W for Woman", "Aurelia", "FabIndia"] }
      ]
    },
    Kids: {
      categories: [
        { title: "Boys Clothing", items: ["T-Shirts", "Shirts", "Jeans", "Ethnic Wear"] },
        { title: "Girls Clothing", items: ["Dresses", "Tops", "Skirts", "Ethnic Wear"] }
      ],
      brands: [
        { title: "Kids Brands", items: ["Mothercare", "Gini & Jony", "Max Kids", "H&M Kids"] }
      ]
    },
    Beauty: {
      categories: [
        { title: "Makeup", items: ["Lipstick", "Foundation", "Kajal", "Mascara"] },
        { title: "Skincare", items: ["Face Wash", "Moisturizer", "Sunscreen", "Serums"] }
      ],
      brands: [
        { title: "Beauty Brands", items: ["Maybelline", "L'Oreal", "Lakme", "Mamaearth"] }
      ]
    },
    "Home & Kitchen": {
      categories: [
        { title: "Bed Linen", items: ["Bedsheets", "Blankets", "Cushions", "Curtains"] },
        { title: "Kitchen", items: ["Cookware", "Serveware", "Storage", "Bottles"] }
      ],
      brands: [
        { title: "Home Brands", items: ["Milton", "Cello", "Prestige", "IKEA"] }
      ]
    }
  };

  const handleSearchSubmit = (e) => {
    if (e) e.preventDefault();
    if (searchQuery.trim()) {
      navigate(`/search?q=${encodeURIComponent(searchQuery.trim())}`);
      setIsOpen(false);
      setActiveDropdown(null);
      setSearchQuery('');
    }
  };

  // --- VOICE SEARCH HANDLER FUNCTION (UPDATED WITH CLICK BLUE & VOLUME SCALE) ---
  const handleVoiceSearch = () => {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
      alert("Browser does not support voice recognition. Try using Chrome or Edge.");
      return;
    }

    const recognition = new SpeechRecognition();
    recognition.lang = 'en-IN';
    recognition.interimResults = true;
    recognition.maxAlternatives = 1;

    setIsListening(true);
    const micBtn = document.querySelector('.voice-search-btn');
    if (micBtn) {
      micBtn.style.color = '#2563eb'; // Click karte hi Blue color
    }

    recognition.onstart = () => {
      setIsListening(true);
    };

    if (window.AudioContext || window.webkitAudioContext) {
      navigator.mediaDevices.getUserMedia({ audio: true }).then((stream) => {
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const analyser = audioCtx.createAnalyser();
        const microphone = audioCtx.createMediaStreamSource(stream);
        microphone.connect(analyser);
        analyser.fftSize = 256;
        const dataArray = new Uint8Array(analyser.frequencyBinCount);

        const checkVolume = () => {
          if (!isListening) return;
          analyser.getByteFrequencyData(dataArray);
          let sum = 0;
          for (let i = 0; i < dataArray.length; i++) {
            sum += dataArray[i];
          }
          let average = sum / dataArray.length;

          if (micBtn) {
            let scaleValue = 1 + (average / 50);
            if (scaleValue > 1.4) scaleValue = 1.4;
            micBtn.style.transform = `scale(${scaleValue})`;
          }
          requestAnimationFrame(checkVolume);
        };
        checkVolume();
      }).catch(err => console.log(err));
    }

    recognition.onresult = (event) => {
      const speechToText = event.results[event.results.length - 1][0].transcript;
      setSearchQuery(speechToText);
      
      if (micBtn) {
        micBtn.innerHTML = '<i class="fa-solid fa-square"></i>'; // Bolte waqt square/stop icon
      }

      if (event.results[0].isFinal && speechToText.trim()) {
        setIsListening(false);
        if (micBtn) micBtn.style.transform = 'scale(1)';
        navigate(`/search?q=${encodeURIComponent(speechToText.trim())}`);
        setIsOpen(false);
        setActiveDropdown(null);
      }
    };

    recognition.onerror = () => {
      setIsListening(false);
      if (micBtn) {
        micBtn.style.color = '';
        micBtn.style.transform = 'scale(1)';
        micBtn.innerHTML = '<i class="fa-solid fa-microphone-lines"></i>';
      }
    };

    recognition.onend = () => {
      setIsListening(false);
      if (micBtn) {
        micBtn.style.color = '';
        micBtn.style.transform = 'scale(1)';
        micBtn.innerHTML = '<i class="fa-solid fa-microphone-lines"></i>';
      }
    };

    recognition.start();
  };

  return (
    <>
      <nav 
        className="navbar" 
        style={{ 
          backgroundColor: navBg, 
          transition: 'background-color 0.5s ease-in-out' 
        }}
      >
        <div className="nav-container">
          
          {/* Logo */}
          <a href="/" onClick={(e) => { e.preventDefault(); navigate('/'); }} className="nav-logo" aria-label="TrendNova home">
            <span className="logo-mark" aria-hidden="true">T</span>
            <span className="logo-wordmark">Trend<span className="logo-highlight">Nova</span></span>
          </a>

          {/* Desktop Mega-Menu Nav Items */}
          <div className="nav-links">
            {Object.keys(categoriesData).map((catName) => (
              <div 
                key={catName} 
                className="nav-item-dropdown"
                onMouseEnter={() => {
                  setActiveDropdown(catName);
                  setActiveTab('categories');
                }}
                onMouseLeave={() => setActiveDropdown(null)}
              >
                <button className={`nav-link-btn ${activeDropdown === catName ? 'active' : ''}`}>
                  <span>{catName}</span>
                  <svg className={`arrow-icon ${activeDropdown === catName ? 'rotate' : ''}`} width="12" height="12" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                    <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2.5" d="M19 9l-7 7-7-7" />
                  </svg>
                </button>

                {activeDropdown === catName && (
                  <div className="mega-dropdown-box">
                    <div className="mega-menu-tabs">
                      <button 
                        className={`tab-btn ${activeTab === 'categories' ? 'active' : ''}`}
                        onClick={() => setActiveTab('categories')}
                      >
                        📂 Categories
                      </button>
                      <button 
                        className={`tab-btn ${activeTab === 'brands' ? 'active' : ''}`}
                        onClick={() => setActiveTab('brands')}
                      >
                        🏷️ Top Brands
                      </button>
                    </div>

                    <div className="mega-menu-content">
                      {activeTab === 'categories' ? (
                        <div className="mega-grid-container">
                          {categoriesData[catName].categories.map((group, idx) => (
                            <div key={idx} className="mega-column">
                              <h4 className="mega-group-title">{group.title}</h4>
                              <ul className="mega-item-list">
                                {group.items.map((item, itemIdx) => (
                                  <li key={itemIdx}>
                                    <span 
                                      onClick={() => {
                                        navigate(`/search?q=${encodeURIComponent(item)}`);
                                        setActiveDropdown(null);
                                      }}
                                      className="mega-item-link"
                                      style={{ cursor: 'pointer' }}
                                    >
                                      {item}
                                    </span>
                                  </li>
                                ))}
                              </ul>
                            </div>
                          ))}
                        </div>
                      ) : (
                        <div className="mega-grid-container brands-grid">
                          {categoriesData[catName].brands.map((group, idx) => (
                            <div key={idx} className="mega-column">
                              <h4 className="mega-group-title brand-title">{group.title}</h4>
                              <ul className="mega-item-list brand-list">
                                {group.items.map((brand, brandIdx) => (
                                  <li key={brandIdx}>
                                    <span 
                                      onClick={() => {
                                        navigate(`/search?q=${encodeURIComponent(brand)}`);
                                        setActiveDropdown(null);
                                      }}
                                      className="mega-item-link brand-link"
                                      style={{ cursor: 'pointer' }}
                                    >
                                      {brand}
                                    </span>
                                  </li>
                                ))}
                              </ul>
                            </div>
                          ))}
                        </div>
                      )}
                    </div>
                  </div>
                )}
              </div>
            ))}
          </div>

          {/* Right Action Icons & Search */}
          <div className="nav-actions">
            <form onSubmit={handleSearchSubmit} className="search-group desktop-search">
              <input
                type="text"
                placeholder="Search for Products, Brands and More"
                className="search-input-field"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
              />
              
              <button 
                type="button" 
                onClick={handleVoiceSearch} 
                className="voice-search-btn"
                title="Search by Voice"
              >
                <i className="fa-solid fa-microphone-lines"></i>
              </button>

              <button type="submit" className="search-submit-btn" aria-label="Search">
                <svg width="16" height="16" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2.5" d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z" />
                </svg>
                <span>Search</span>
              </button>
            </form>

            <button
              onClick={() => setDarkMode(!darkMode)}
              className="theme-icon-btn"
              title={darkMode ? "Switch to Light Mode" : "Switch to Dark Mode"}
            >
              {darkMode ? '☀︎' : '⏾'}
            </button>

            {/* --- SIGN IN / PROFILE CONDITION WITH DARK MODE FIX --- */}
            {token ? (
              <div className="user-profile-container" style={{ display: 'flex', alignItems: 'center', gap: '12px' }}>
                {role === 'admin' && (
                  <Link 
                    to="/admin-dashboard"
                    className="admin-link"
                    style={{  
                      textDecoration: 'none', 
                      padding: '6px 12px',
                      backgroundColor: darkMode ? '#1e293b' : '#fee2e2',
                      color: darkMode ? '#f472b6' : '#dc2626',
                      borderRadius: '20px',
                      fontWeight: '600',
                      border: darkMode ? '1px solid #334155' : '1px solid #c9bdbd',
                      display: 'flex',
                      alignItems: 'center',
                      gap: '6px'
                    }}
                  >
                    <i className="fa-solid fa-unlock"></i>
                    <span>Admin</span>
                  </Link>
                )}

                <button onClick={handleLogout} className="logout-btn" style={{ padding: '5px 10px', background: '#333', color: '#fff', border: 'none', cursor: 'pointer', borderRadius: '4px' }}>
                  Logout
                </button>
              </div>
            ) : (
              <span onClick={handleGoogleSignIn} className="action-item signin-desktop" style={{ cursor: 'pointer', display: 'flex', alignItems: 'center', gap: '6px' }}>
                <svg width="18" height="18" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M16 7a4 4 0 11-8 0 4 4 0 018 0zM12 14a7 7 0 00-7 7h14a7 7 0 00-7-7z" />
                </svg>
                <span>Sign In</span>
              </span>
            )}

            <span onClick={() => navigate('/cart')} className="action-item cart-wrapper" title="View Cart" style={{ cursor: 'pointer' }}>
              <svg width="20" height="20" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M3 3h2l.4 2M7 13h10l4-8H5.4M7 13L5.4 5M7 13l-2.293 2.293c-.63.63-.184 1.707.707 1.707H17m0 0a2 2 0 100 4 2 2 0 000-4zm-8 2a2 2 0 11-4 0 2 2 0 014 0z" />
              </svg>
              <span className="cart-badge">0</span>
            </span>

            <button
              onClick={() => setIsOpen(!isOpen)}
              className="mobile-toggle"
              aria-label="Toggle Menu"
            >
              <svg width="22" height="22" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                {isOpen ? (
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M6 18L18 6M6 6l12 12" />
                ) : (
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M4 6h16M4 12h16M4 18h16" />
                )}
              </svg>
            </button>
          </div>

        </div>

        {/* Mobile Search Row */}
        <div className="mobile-search-always-bar" style={{ backgroundColor: navBg, transition: 'background-color 0.5s ease-in-out' }}>
          <form onSubmit={handleSearchSubmit} className="search-group mobile-always-search">
            <input
              type="text"
              placeholder="Search products & brands..."
              className="search-input-field"
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
            />
            
            <button type="button" onClick={handleVoiceSearch} className="voice-search-btn" title="Search by Voice">
              <i className="fa-solid fa-microphone"></i>
            </button>

            <button type="submit" className="search-submit-btn" aria-label="Search">
              <svg width="15" height="15" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2.5" d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z" />
              </svg>
            </button>
          </form>
        </div>

        {/* Mobile Accordion Menu */}
        <div className={`mobile-menu ${isOpen ? 'open' : ''}`}>
          <div className="mobile-nav-links">
            {Object.keys(categoriesData).map((catName) => (
              <details key={catName} className="mobile-category-block">
                <summary className="mobile-cat-title">{catName}</summary>
                <div className="mobile-accordion-body">
                  <div className="mobile-sub-section">
                    <div className="mobile-sub-heading">Categories</div>
                    {categoriesData[catName].categories.map((group, idx) => (
                      <div key={idx} className="mobile-group-box">
                        <strong>{group.title}</strong>
                        <div className="mobile-items-wrap">
                          {group.items.map((item, itemIdx) => (
                            <span 
                              key={itemIdx} 
                              onClick={() => {
                                navigate(`/search?q=${encodeURIComponent(item)}`);
                                setIsOpen(false);
                              }}
                              className="mobile-sub-link"
                              style={{ cursor: 'pointer' }}
                            >
                              {item}
                            </span>
                          ))}
                        </div>
                      </div>
                    ))}
                  </div>
                </div>
              </details>
            ))}
          </div>

          {/* --- MOBILE AUTH BUTTONS (ADMIN & LOGOUT OR SIGN IN) --- */}
          {token ? (
            <div style={{ marginTop: '18px', display: 'flex', flexDirection: 'column', gap: '10px' }}>
              {role === 'admin' && (
                <Link 
                  to="/admin-dashboard" 
                  onClick={() => setIsOpen(false)}
                  style={{
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    gap: '8px',
                    width: '100%',
                    minHeight: '44px',
                    padding: '10px 16px',
                    borderRadius: '10px',
                    backgroundColor: '#fee2e2',
                    color: '#dc2626',
                    fontSize: '13.5px',
                    fontWeight: '750',
                    textDecoration: 'none'
                  }}
                >
                  <i className="fa-solid fa-unlock"></i>
                  <span>Admin Dashboard</span>
                </Link>
              )}

              <button 
                onClick={() => {
                  setIsOpen(false);
                  handleLogout();
                }} 
                style={{
                  display: 'flex',
                  alignItems: 'center',
                  justifyContent: 'center',
                  gap: '8px',
                  width: '100%',
                  minHeight: '44px',
                  padding: '10px 16px',
                  borderRadius: '10px',
                  background: '#333',
                  color: '#fff',
                  border: 'none',
                  fontSize: '13.5px',
                  fontWeight: '750',
                  cursor: 'pointer'
                }}
              >
                <i className="fa-solid fa-right-from-bracket"></i>
                <span>Logout</span>
              </button>
            </div>
          ) : (
            <span onClick={handleGoogleSignIn} className="mobile-signin-btn" style={{ cursor: 'pointer', display: 'flex' }}>
              <svg width="18" height="18" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M16 7a4 4 0 11-8 0 4 4 0 018 0zM12 14a7 7 0 00-7 7h14a7 7 0 00-7-7z" />
              </svg>
              <span>Sign In with Google</span>
            </span>
          )}
        </div>

        {/* Sub-Categories Bar */}
        {location.pathname === '/' && (
          <div className={`sub-categories-bar ${darkMode ? 'dark-sub-bar' : ''} ${isScrolled ? 'scrolled-sub-bar' : ''}`}>
            <div className="sub-categories-track">
              {subCategoriesList.map((cat, index) => (
                <span key={index} onClick={() => navigate(cat.link)} className="sub-cat-item" style={{ cursor: 'pointer' }}>
                  <div className={`sub-cat-icon-box ${darkMode ? 'dark-icon-box' : ''}`}>
                    <span className="sub-cat-emoji">{cat.icon}</span>
                  </div>
                  <span className={`sub-cat-name ${darkMode ? 'dark-text' : ''}`}>{cat.title}</span>
                </span>
              ))}
            </div>
          </div>
        )}
      </nav>

      {/* Embedded Styling */}
      <style>{`
        * { box-sizing: border-box; }
        body { padding-top: 80px; }
        @media (max-width: 980px) { body { padding-top: 125px; } }

        .navbar {
          position: fixed;
          top: 0;
          left: 0;
          right: 0;
          width: 100%;
          color: #0f172a;
          box-shadow: 0 2px 14px rgba(15, 23, 42, 0.07);
          z-index: 1000;
          font-family: Inter, sans-serif;
        }
        .nav-container {
          width: 100%;
          max-width: 1600px;
          min-height: 60px;
          margin: 0 auto;
          padding: 6px 28px;
          display: flex;
          align-items: center;
          gap: 24px;
        }
        .nav-logo {
          display: inline-flex;
          align-items: center;
          font-size: 23px;
          font-weight: 850;
          color: #111827;
          text-decoration: none;
        }
        .logo-highlight { color: #db2777; }
        .nav-links { display: flex; flex: 1; align-items: center; justify-content: center; gap: 20px; }
        .nav-item-dropdown { position: relative; display: flex; align-items: center; }
        .nav-link-btn { background: transparent; border: 0; font-weight: 650; cursor: pointer; color: #334155; }
        .nav-link-btn:hover { color: #db2777; }
        
        .mega-dropdown-box {
          position: fixed;
          top: 60px;
          left: 50%;
          transform: translateX(-50%);
          width: min(92vw, 1360px);
          background: #fff;
          border: 1px solid #e2e8f0;
          border-radius: 14px;
          box-shadow: 0 22px 55px rgba(15, 23, 42, 0.16);
          z-index: 1001;
        }
        .mega-menu-tabs { display: flex; gap: 10px; padding: 14px 22px; background: #f8fafc; border-bottom: 1px solid #e2e8f0; }
        .tab-btn { padding: 8px 15px; border: 1px solid #cbd5e1; border-radius: 999px; background: #fff; cursor: pointer; font-weight: 650; }
        .tab-btn.active { color: #fff; background: #db2777; border-color: #db2777; }
        .mega-menu-content { padding: 22px; }
        .mega-grid-container { display: grid; grid-template-columns: repeat(auto-fit, minmax(175px, 1fr)); gap: 24px; }
